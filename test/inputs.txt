# A send with a value for the event's data

action tell = send(list, removed, self)

machine todoItem {
  initial state active {
    rename => send(list, renamed, event.data) => active
    touch => send(parent, touched, context.id) => send(parent, summary, :summarise) => active
    plain => send(parent, forwarded) => active
  }
}

# Symbols given in a spawn's block, beside inputs

machine todoList {
  initial state ready {
    add => append(todos, spawn(todoItem) {
      title = event.data
      :canDelete = :canDeleteTodo
      :log = log
    }) => ready
    one => assign(todo, spawn(todoItem) { :log = log }) => ready
  }
}
